\makesheethead{問題}

\problem[サイズの判定]
$A$ は $3\times2$、$B$ は $2\times4$ 行列とする。$AB$ と $BA$ は定義できるか。定義できる場合はサイズも答えよ。
\workspace{38mm}

\problem[積と非可換性]
$A=\begin{pmatrix}1&2\\0&1\end{pmatrix}$、$B=\begin{pmatrix}2&0\\1&1\end{pmatrix}$ とする。$AB,BA$ と $A(2,-1)^{\mathsf T}$ を求めよ。
\workspace{38mm}
\newpage

\problem[回転]
$R=\begin{pmatrix}0&-1\\1&0\end{pmatrix}$ とする。$R(2,1)^{\mathsf T}$、$R^2$ を求め、それぞれ図形的な意味を説明せよ。
\workspace{38mm}

\problem[センサの出力]
$y_1=x_1+x_2,\ y_2=x_1-x_2$ を行列で表せ。$x_1=5,\ x_2=2$ のとき出力を求めよ。
\workspace{38mm}
